Подготовка компьютера

Один скрипт ставит Git, Node.js LTS (node, npm, npx) и VS Code, прописывает PATH и настраивает Git. После него откройте новый терминал, и всё работает.

  1. Нажмите Win, наберите PowerShell, откройте.
  2. Скопируйте команду, вставьте правой кнопкой мыши, нажмите Enter.
  3. Разрешите установку, если Windows спросит. Введите имя и email от GitHub.
  4. Закройте окно, откройте новое и проверьте: node -v
irm https://monforje.github.io/lendingi-2026/setup/win.txt | iex

Нужен Windows 10 или 11. Скрипт использует winget, а если его нет, качает установщики с официальных сайтов. Ошибка «выполнение сценариев отключено» больше не появится: скрипт разрешает npm запускаться в PowerShell.

# Lending kit setup for Windows 10/11: Git, Node.js LTS, VS Code, PATH.
# Run in PowerShell (not as admin needed, Windows will ask for permission itself):
#   irm https://monforje.github.io/lendingi-2026/setup/win.txt | iex

$ErrorActionPreference = 'Stop'
try { [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12 } catch {}

function Say($t) { Write-Host "`n== $t" -ForegroundColor Cyan }
function Ok($t)  { Write-Host "   OK  $t" -ForegroundColor Green }
function Warn($t){ Write-Host "   !!  $t" -ForegroundColor Yellow }
function Has($c) { [bool](Get-Command $c -ErrorAction SilentlyContinue) }
function Refresh-Path {
  $m = [Environment]::GetEnvironmentVariable('Path','Machine')
  $u = [Environment]::GetEnvironmentVariable('Path','User')
  $env:Path = ($m + ';' + $u)
}
function Add-UserPath($dir) {
  if (-not (Test-Path $dir)) { return }
  $u = [Environment]::GetEnvironmentVariable('Path','User'); if (-not $u) { $u = '' }
  if (($u -split ';') -notcontains $dir) {
    [Environment]::SetEnvironmentVariable('Path', ($u.TrimEnd(';') + ';' + $dir).TrimStart(';'), 'User')
    Ok "PATH += $dir"
  }
}
$HasWinget = Has winget
function WG($id) {
  winget install -e --id $id --silent --accept-source-agreements --accept-package-agreements
  if ($LASTEXITCODE -ne 0 -and $LASTEXITCODE -ne -1978335189) { throw "winget $id exit $LASTEXITCODE" }
}
function Download($url, $out) {
  Write-Host "   download $url"
  Invoke-WebRequest -UseBasicParsing -Uri $url -OutFile $out
}

Write-Host "Lending kit setup: Git + Node.js LTS + VS Code" -ForegroundColor Cyan
if (-not $HasWinget) { Warn "winget not found, using direct downloads" }

# 1. Git
Say "Git"
Refresh-Path
if (Has git) { Ok (git --version) } else {
  try {
    if ($HasWinget) { WG 'Git.Git' } else { throw 'no winget' }
  } catch {
    $rel = Invoke-RestMethod 'https://api.github.com/repos/git-for-windows/git/releases/latest'
    $arch = if ($env:PROCESSOR_ARCHITECTURE -eq 'ARM64') { 'arm64' } else { '64-bit' }
    $asset = $rel.assets | Where-Object { $_.name -match "^Git-.*-$arch\.exe$" } | Select-Object -First 1
    $exe = Join-Path $env:TEMP $asset.name
    Download $asset.browser_download_url $exe
    Start-Process $exe -ArgumentList '/VERYSILENT','/NORESTART','/SP-' -Verb RunAs -Wait
  }
  Add-UserPath "$env:ProgramFiles\Git\cmd"
  Refresh-Path
  if (Has git) { Ok (git --version) } else { Warn "Git installed, open a new terminal" }
}

# 2. Node.js LTS (node + npm + npx)
Say "Node.js LTS"
if (Has node) { Ok ("node " + (node -v)) } else {
  try {
    if ($HasWinget) { WG 'OpenJS.NodeJS.LTS' } else { throw 'no winget' }
  } catch {
    $lts = (Invoke-RestMethod 'https://nodejs.org/dist/index.json' | Where-Object { $_.lts } | Select-Object -First 1).version
    $arch = if ($env:PROCESSOR_ARCHITECTURE -eq 'ARM64') { 'arm64' } else { 'x64' }
    $msi = Join-Path $env:TEMP "node-$lts-$arch.msi"
    Download "https://nodejs.org/dist/$lts/node-$lts-$arch.msi" $msi
    Start-Process msiexec.exe -ArgumentList '/i', "`"$msi`"", '/qn', '/norestart' -Verb RunAs -Wait
  }
  Refresh-Path
}
Add-UserPath "$env:ProgramFiles\nodejs"
Add-UserPath "$env:APPDATA\npm"
New-Item -ItemType Directory -Force "$env:APPDATA\npm" | Out-Null
Refresh-Path
if (Has node) { Ok ("node " + (node -v) + ", npm " + (npm -v)) } else { Warn "Node installed, open a new terminal" }

# npm scripts (*.ps1) are blocked by default on Windows. Allow local scripts for this user only.
try {
  if ((Get-ExecutionPolicy -Scope CurrentUser) -in 'Undefined','Restricted','AllSigned') {
    Set-ExecutionPolicy -Scope CurrentUser RemoteSigned -Force
    Ok "ExecutionPolicy CurrentUser = RemoteSigned (npm, npx work in PowerShell)"
  }
} catch { Warn "Could not change ExecutionPolicy: $($_.Exception.Message)" }

# 3. VS Code
Say "VS Code"
if (Has code) { Ok "VS Code already installed" } else {
  try {
    if ($HasWinget) { WG 'Microsoft.VisualStudioCode' } else { throw 'no winget' }
  } catch {
    $exe = Join-Path $env:TEMP 'VSCodeUserSetup.exe'

    Download "https://update.code.visualstudio.com/latest/$a/stable" $exe
    Start-Process $exe -ArgumentList '/VERYSILENT','/NORESTART','/MERGETASKS=!runcode,addcontextmenufiles,addcontextmenufolders,addtopath' -Wait
  }
  Add-UserPath "$env:LOCALAPPDATA\Programs\Microsoft VS Code\bin"
  Refresh-Path
  if (Has code) { Ok "VS Code installed" } else { Warn "VS Code installed, open a new terminal" }
}

# 4. Git settings
Say "Git settings"
if (Has git) {
  git config --global init.defaultBranch main
  git config --global core.autocrlf true
  if (-not (git config --global user.name)) {
    $n = Read-Host "   Your name for commits (Enter = skip)"
    if ($n) { git config --global user.name "$n" }
  }
  if (-not (git config --global user.email)) {
    $e = Read-Host "   Email from GitHub (Enter = skip)"
    if ($e) { git config --global user.email "$e" }
  }
  Ok "branch main, user: $(git config --global user.name) <$(git config --global user.email)>"
}

Say "Done"

}

#!/usr/bin/env bash
# Lending kit setup for macOS and Linux: Git, Node.js LTS (via nvm), PATH in your shell profile.
# Run in Terminal:
#   curl -fsSL https://monforje.github.io/lendingi-2026/setup/install.sh | bash
set -euo pipefail

say()  { printf '\n\033[1;36m== %s\033[0m\n' "$1"; }
ok()   { printf '   \033[32mOK\033[0m  %s\n' "$1"; }
warn() { printf '   \033[33m!!\033[0m  %s\n' "$1"; }
has()  { command -v "$1" >/dev/null 2>&1; }
ask()  { local a=""; if [ -r /dev/tty ]; then read -r -p "   $1" a </dev/tty || true; fi; printf '%s' "$a"; }

OS="$(uname -s)"
SUDO=""; if [ "$(id -u)" -ne 0 ] && has sudo; then SUDO="sudo"; fi

# Shell profile where PATH is written
case "${SHELL:-}" in
  */zsh)  RC="$HOME/.zshrc" ;;
  */bash) if [ "$OS" = "Darwin" ]; then RC="$HOME/.bash_profile"; else RC="$HOME/.bashrc"; fi ;;
  *)      RC="$HOME/.profile" ;;
esac
touch "$RC"
printf '\033[1;36mLending kit setup: Git + Node.js LTS. Profile: %s\033[0m\n' "$RC"

# 1. Git (+ curl)
say "Git"
if [ "$OS" = "Darwin" ]; then
  if xcode-select -p >/dev/null 2>&1; then ok "$(git --version)"; else
    warn "A window will open: press Install (Command Line Tools, 5-15 min)"
    xcode-select --install || true
    for _ in $(seq 1 180); do xcode-select -p >/dev/null 2>&1 && break; sleep 10; done
    xcode-select -p >/dev/null 2>&1 && ok "$(git --version)" || { warn "Finish Command Line Tools install and run this script again"; exit 1; }
  fi
else
  if has git && has curl; then ok "$(git --version)"; else
    if   has apt-get; then $SUDO apt-get update -y && $SUDO apt-get install -y git curl ca-certificates
    elif has dnf;     then $SUDO dnf install -y git curl
    elif has yum;     then $SUDO yum install -y git curl
    elif has pacman;  then $SUDO pacman -Sy --noconfirm git curl
    elif has zypper;  then $SUDO zypper --non-interactive install git curl
    elif has apk;     then $SUDO apk add git curl bash
    else warn "Unknown package manager: install git and curl manually"; exit 1; fi
    ok "$(git --version)"
  fi
fi

# 2. Node.js LTS through nvm (no sudo, easy to update)
say "Node.js LTS"
export NVM_DIR="$HOME/.nvm"
if [ ! -s "$NVM_DIR/nvm.sh" ]; then
  NVM_VER="$(curl -fsSL https://api.github.com/repos/nvm-sh/nvm/releases/latest 2>/dev/null | sed -n 's/.*"tag_name": *"\(v[^"]*\)".*/\1/p' | head -1 || true)"
  NVM_VER="${NVM_VER:-v0.40.3}"
  curl -fsSL "https://raw.githubusercontent.com/nvm-sh/nvm/$NVM_VER/install.sh" | PROFILE="$RC" bash >/dev/null
  ok "nvm $NVM_VER"
fi
# nvm.sh is not compatible with `set -u`
set +u
# shellcheck disable=SC1091
. "$NVM_DIR/nvm.sh"
nvm install --lts || { warn "Could not download Node.js from nodejs.org. Check internet/VPN and run the script again"; exit 1; }
nvm alias default 'lts/*' >/dev/null
nvm use default >/dev/null
set -u
ok "node $(node -v), npm $(npm -v)"

# PATH block in profile (nvm usually writes it, this makes sure)
if ! grep -q 'NVM_DIR' "$RC"; then
  {
    echo ''
    echo '# nvm + Node.js (lending kit)'
    echo 'export NVM_DIR="$HOME/.nvm"'
    echo '[ -s "$NVM_DIR/nvm.sh" ] && . "$NVM_DIR/nvm.sh"'
  } >> "$RC"
fi
if ! grep -q '.local/bin' "$RC"; then
  echo 'export PATH="$HOME/.local/bin:$PATH"' >> "$RC"
fi
ok "PATH written to $RC"

# 3. VS Code (macOS with Homebrew only; otherwise link)
say "VS Code"
if has code; then ok "VS Code found"
elif [ "$OS" = "Darwin" ] && has brew; then brew install --cask visual-studio-code && ok "VS Code installed"
else warn "Download VS Code: https://code.visualstudio.com/download"; fi

# 4. Git settings
say "Git settings"
git config --global init.defaultBranch main
git config --global core.autocrlf input
if [ -z "$(git config --global user.name || true)" ]; then
  n="$(ask 'Your name for commits (Enter = skip): ')"; [ -n "$n" ] && git config --global user.name "$n"
fi
if [ -z "$(git config --global user.email || true)" ]; then
  e="$(ask 'Email from GitHub (Enter = skip): ')"; [ -n "$e" ] && git config --global user.email "$e"
fi
ok "branch main, user: $(git config --global user.name || true) <$(git config --global user.email || true)>"

say "Done"
printf '\033[32mOpen a NEW terminal window (or run: source %s). Then check: node -v; npm -v; git --version\033[0m\n' "$RC"

  2. Вставьте команду (Ctrl + Shift + V), нажмите Enter, введите пароль для sudo.
  3. Откройте новый терминал и проверьте: node -v
